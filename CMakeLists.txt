cmake_minimum_required(VERSION 3.16)
project(esp32_hands_on_academy LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

set(ACADEMY_LESSON "" CACHE STRING "Optional lesson slug for the shared device lab")
if(ACADEMY_LESSON)
    if(NOT ACADEMY_LESSON MATCHES "^[0-9][0-9]-[a-z0-9-]+$")
        message(FATAL_ERROR "Invalid lesson slug")
    endif()
    set(LAB_VARIANT "starter" CACHE STRING "starter, challenge, or solution")
    if(NOT LAB_VARIANT MATCHES "^(starter|challenge|solution)$")
        message(FATAL_ERROR "Invalid LAB_VARIANT")
    endif()
    set(LESSON_DIR "${CMAKE_CURRENT_SOURCE_DIR}/lessons/${ACADEMY_LESSON}")
    set(LAB_SOURCE_DIR "" CACHE PATH "Notebook controller directory")
    if(LAB_SOURCE_DIR)
        set(CONTROLLER_DIR "${LAB_SOURCE_DIR}")
    else()
        set(CONTROLLER_DIR "${LESSON_DIR}/${LAB_VARIANT}")
    endif()
    add_library(controller "${CONTROLLER_DIR}/controller.cpp")
    target_include_directories(controller PUBLIC include "${CONTROLLER_DIR}")
    if(MSVC)
        target_compile_options(controller PRIVATE /W4)
    else()
        target_compile_options(controller PRIVATE -Wall -Wextra -Wpedantic)
    endif()
    add_executable(simulator simulator/device_main.cpp)
    target_link_libraries(simulator PRIVATE controller)
    add_executable(controller_tests "${LESSON_DIR}/tests/controller_tests.cpp")
    target_link_libraries(controller_tests PRIVATE controller)
    enable_testing()
    add_test(NAME lesson_behavior COMMAND controller_tests exercise)
    return()
endif()

# Default builds compile only the learner's code. Solutions are opt-in.
set(LAB_VARIANT "starter" CACHE STRING "starter, challenge, or solution")
set_property(CACHE LAB_VARIANT PROPERTY STRINGS starter challenge solution)
if(NOT LAB_VARIANT MATCHES "^(starter|challenge|solution)$")
    message(FATAL_ERROR "Invalid LAB_VARIANT")
endif()
set(LESSON_DIR "${CMAKE_CURRENT_SOURCE_DIR}/lessons/01-smart-button")
set(LAB_SOURCE_DIR "" CACHE PATH "Optional notebook-generated controller directory")
if(LAB_SOURCE_DIR)
    set(CONTROLLER_DIR "${LAB_SOURCE_DIR}")
else()
    set(CONTROLLER_DIR "${LESSON_DIR}/${LAB_VARIANT}")
endif()
add_library(controller "${CONTROLLER_DIR}/controller.cpp")
target_include_directories(controller PUBLIC include "${CONTROLLER_DIR}")
if(MSVC)
    target_compile_options(controller PRIVATE /W4)
else()
    target_compile_options(controller PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(simulator simulator/main.cpp)
target_include_directories(simulator PRIVATE simulator)
target_link_libraries(simulator PRIVATE controller)
add_executable(controller_tests "${LESSON_DIR}/tests/controller_tests.cpp")
target_include_directories(controller_tests PRIVATE simulator)
target_link_libraries(controller_tests PRIVATE controller)
enable_testing()
if(LAB_VARIANT STREQUAL "starter")
    set(DEFAULT_PROFILE "guided")
else()
    set(DEFAULT_PROFILE "exercise")
endif()
set(LAB_PROFILE "${DEFAULT_PROFILE}" CACHE STRING "guided or exercise")
if(NOT LAB_PROFILE MATCHES "^(guided|exercise)$")
    message(FATAL_ERROR "Invalid LAB_PROFILE")
endif()
add_test(NAME button_behavior COMMAND controller_tests "${LAB_PROFILE}")
